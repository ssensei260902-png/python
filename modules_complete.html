Module 4: Functions and Scope

Learn to write reusable, modular code with functions - the building blocks of professional programming!

4.1 What Are Functions?

Functions are reusable blocks of code that perform specific tasks. Think of them as mini-programs within your program!

💡 Why Use Functions?

  • Reusability: Write once, use many times
  • Organization: Break complex problems into smaller pieces
  • Maintainability: Fix bugs in one place
  • Abstraction: Hide complexity behind simple interfaces

Example: Basic Function

# Defining a function
def greet():
    print("Hello, World!")
    print("Welcome to Python!")

# Calling the function
greet()
greet()  # Can call multiple times!

# Function with parameters
def greet_person(name):
    print(f"Hello, {name}!")
    print(f"Nice to meet you, {name}!")

greet_person("Alice")
greet_person("Bob")

# Function with multiple parameters
def introduce(name, age, city):
    print(f"My name is {name}")
    print(f"I am {age} years old")
    print(f"I live in {city}")

introduce("Alice", 25, "New York")

4.2 Return Values

Functions can return values using the return statement. This allows functions to compute and send back results.

Example: Functions with Return Values

# Function that returns a value
def add(a, b):
    result = a + b
    return result

sum_result = add(5, 3)
print(f"5 + 3 = {sum_result}")

# Function with multiple return values (returns tuple)
def calculate(a, b):
    sum_val = a + b
    diff = a - b
    product = a * b
    quotient = a / b if b != 0 else None
    return sum_val, diff, product, quotient

# Unpack the results
s, d, p, q = calculate(10, 5)
print(f"Sum: {s}, Diff: {d}, Product: {p}, Quotient: {q}")

# Real-world example: Temperature converter
def celsius_to_fahrenheit(celsius):
    fahrenheit = (celsius * 9/5) + 32
    return fahrenheit

temp_c = 25
temp_f = celsius_to_fahrenheit(temp_c)
print(f"{temp_c}°C = {temp_f}°F")

# Function that returns boolean
def is_even(number):
    return number % 2 == 0

print(f"Is 10 even? {is_even(10)}")
print(f"Is 7 even? {is_even(7)}")

4.3 Default Arguments and Keyword Arguments

Python allows you to set default values for parameters and call functions using keyword arguments.

# Default arguments
def greet(name, greeting="Hello"):
    print(f"{greeting}, {name}!")

greet("Alice")              # Uses default greeting
greet("Bob", "Hi")         # Uses custom greeting
greet("Charlie", "Hey")    # Uses custom greeting

# Keyword arguments (can be in any order!)
def create_profile(name, age, city, country):
    print(f"{name}, {age} years old")
    print(f"Lives in {city}, {country}")

# Using keyword arguments
create_profile(name="Alice", age=25, city="NYC", country="USA")
create_profile(age=30, country="UK", name="Bob", city="London")

# Mix positional and keyword arguments
create_profile("Charlie", age=28, city="Paris", country="France")

# *args - Variable number of arguments
def sum_all(*numbers):
    total = 0
    for num in numbers:
        total += num
    return total

print(sum_all(1, 2, 3))           # 6
print(sum_all(10, 20, 30, 40))   # 100

# **kwargs - Variable number of keyword arguments
def print_info(**info):
    for key, value in info.items():
        print(f"{key}: {value}")

print_info(name="Alice", age=25, city="NYC", job="Engineer")

4.4 Scope and Lifetime of Variables

Scope determines where variables can be accessed. Python has local, enclosing, global, and built-in scopes (LEGB rule).

# Global variable
global_var = "I'm global"

def my_function():
    # Local variable
    local_var = "I'm local"
    print(global_var)  # Can access global
    print(local_var)   # Can access local

my_function()
# print(local_var)  # Error! Local var doesn't exist here

# Modifying global variables
count = 0

def increment():
    global count  # Declare we're using global variable
    count += 1
    print(f"Count: {count}")

increment()  # Count: 1
increment()  # Count: 2
increment()  # Count: 3

# Best practice: Avoid global variables, use return instead
def increment_better(current_count):
    return current_count + 1

my_count = 0
my_count = increment_better(my_count)
my_count = increment_better(my_count)
print(f"My count: {my_count}")

# Enclosing scope (nested functions)
def outer():
    x = "outer"

    def inner():
        x = "inner"
        print(f"Inner x: {x}")

    inner()
    print(f"Outer x: {x}")

outer()

4.5 Lambda Functions

Lambda functions are small, anonymous functions defined with the lambda keyword. Perfect for simple operations!

# Regular function
def square(x):
    return x ** 2

# Lambda function (one-liner!)
square_lambda = lambda x: x ** 2

print(square(5))         # 25
print(square_lambda(5))  # 25

# Lambda with multiple arguments
add = lambda a, b: a + b
print(add(3, 7))  # 10

# Using lambda with built-in functions
numbers = [1, 2, 3, 4, 5]

# Map: Apply function to each element
squared = list(map(lambda x: x ** 2, numbers))
print(f"Squared: {squared}")

# Filter: Keep elements that match condition
evens = list(filter(lambda x: x % 2 == 0, numbers))
print(f"Evens: {evens}")

# Sorted with custom key
students = [
    {"name": "Alice", "grade": 85},
    {"name": "Bob", "grade": 92},
    {"name": "Charlie", "grade": 78}
]

sorted_students = sorted(students, key=lambda s: s["grade"], reverse=True)
for student in sorted_students:
    print(f"{student['name']}: {student['grade']}")

4.6 Recursion

Recursion is when a function calls itself. Powerful for solving problems that can be broken into smaller, similar problems!

# Factorial using recursion
def factorial(n):
    # Base case: factorial of 0 or 1 is 1
    if n <= 1:
        return 1
    # Recursive case
    return n * factorial(n - 1)

print(f"5! = {factorial(5)}")  # 120

# Fibonacci sequence
def fibonacci(n):
    if n <= 1:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

print("Fibonacci sequence:")
for i in range(10):
    print(fibonacci(i), end=" ")
print()

# Countdown using recursion
def countdown(n):
    if n <= 0:
        print("Blast off!")
    else:
        print(n)
        countdown(n - 1)

countdown(5)

# Real-world: Calculate power
def power(base, exp):
    if exp == 0:
        return 1
    return base * power(base, exp - 1)

print(f"2^8 = {power(2, 8)}")

5.1 Introduction to OOP

Object-Oriented Programming (OOP) organizes code around objects that combine data and behavior. It's like creating blueprints (classes) and building things from them (objects).

💡 Four Pillars of OOP

  • Encapsulation: Bundling data and methods together
  • Inheritance: Creating new classes from existing ones
  • Polymorphism: Same interface, different implementations
  • Abstraction: Hiding complex implementation details

Example: Your First Class

# Defining a class
class Dog:
    # Constructor (initializer)
    def __init__(self, name, age):
        self.name = name  # Instance variable
        self.age = age

    # Instance method
    def bark(self):
        print(f"{self.name} says: Woof!")

    def get_info(self):
        print(f"{self.name} is {self.age} years old")

# Creating objects (instances)
dog1 = Dog("Buddy", 3)
dog2 = Dog("Max", 5)

# Using objects
dog1.bark()         # Buddy says: Woof!
dog2.bark()         # Max says: Woof!
dog1.get_info()     # Buddy is 3 years old
dog2.get_info()     # Max is 5 years old

# Accessing attributes
print(f"Dog 1's name: {dog1.name}")
print(f"Dog 2's age: {dog2.age}")

5.2 Class vs Instance Variables

Class variables are shared by all instances. Instance variables are unique to each object.

class BankAccount:
    # Class variable (shared by all instances)
    bank_name = "Python Bank"
    total_accounts = 0

    def __init__(self, owner, balance=0):
        # Instance variables (unique to each object)
        self.owner = owner
        self.balance = balance
        BankAccount.total_accounts += 1

    def deposit(self, amount):
        self.balance += amount
        print(f"Deposited ${amount}. New balance: ${self.balance}")

    def withdraw(self, amount):
        if amount > self.balance:
            print("Insufficient funds!")
        else:
            self.balance -= amount
            print(f"Withdrew ${amount}. New balance: ${self.balance}")

    def get_balance(self):
        print(f"{self.owner}'s balance: ${self.balance}")

# Create accounts
acc1 = BankAccount("Alice", 1000)
acc2 = BankAccount("Bob", 500)

# Use instance methods
acc1.deposit(500)
acc1.withdraw(200)
acc1.get_balance()

acc2.deposit(1000)
acc2.get_balance()

# Access class variable
print(f"Bank: {BankAccount.bank_name}")
print(f"Total accounts: {BankAccount.total_accounts}")

5.3 Inheritance

Inheritance allows you to create new classes based on existing ones, inheriting their attributes and methods.

# Parent class (Base class)
class Animal:
    def __init__(self, name, species):
        self.name = name
        self.species = species

    def make_sound(self):
        print(f"{self.name} makes a sound")

    def info(self):
        print(f"{self.name} is a {self.species}")

# Child class (Derived class)
class Dog(Animal):
    def __init__(self, name, breed):
        super().__init__(name, "Dog")  # Call parent constructor
        self.breed = breed

    # Override parent method
    def make_sound(self):
        print(f"{self.name} says: Woof!")

    # New method specific to Dog
    def fetch(self):
        print(f"{self.name} is fetching the ball!")

class Cat(Animal):
    def __init__(self, name, color):
        super().__init__(name, "Cat")
        self.color = color

    def make_sound(self):
        print(f"{self.name} says: Meow!")

    def climb(self):
        print(f"{self.name} is climbing a tree!")

# Create instances
dog = Dog("Buddy", "Golden Retriever")
cat = Cat("Whiskers", "Orange")

# Use inherited and new methods
dog.info()          # Inherited from Animal
dog.make_sound()    # Overridden in Dog
dog.fetch()         # Specific to Dog

cat.info()
cat.make_sound()
cat.climb()

5.4 Special Methods (Magic Methods)

Special methods (dunder methods - double underscore) allow you to define behavior for built-in operations.

class Book:
    def __init__(self, title, author, pages):
        self.title = title
        self.author = author
        self.pages = pages

    # String representation for users
    def __str__(self):
        return f"'{self.title}' by {self.author}"

    # String representation for developers
    def __repr__(self):
        return f"Book('{self.title}', '{self.author}', {self.pages})"

    # Length of book
    def __len__(self):
        return self.pages

    # Comparison operators
    def __eq__(self, other):
        return self.pages == other.pages

    def __lt__(self, other):
        return self.pages < other.pages

book1 = Book("Python Mastery", "John Doe", 300)
book2 = Book("Advanced Python", "Jane Smith", 450)

print(str(book1))        # Uses __str__
print(repr(book1))       # Uses __repr__
print(f"Pages: {len(book1)}")  # Uses __len__

if book1 < book2:
    print(f"{book1.title} has fewer pages")

# More magic methods example
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __add__(self, other):
        return Vector(self.x + other.x, self.y + other.y)

    def __str__(self):
        return f"Vector({self.x}, {self.y})"

v1 = Vector(2, 3)
v2 = Vector(5, 7)
v3 = v1 + v2  # Uses __add__
print(v3)     # Vector(7, 10)

5.5 Properties and Encapsulation

Use properties to control access to attributes and maintain encapsulation.

class Temperature:
    def __init__(self, celsius):
        self._celsius = celsius  # Protected attribute

    @property
    def celsius(self):
        return self._celsius

    @celsius.setter
    def celsius(self, value):
        if value < -273.15:
            raise ValueError("Temperature below absolute zero!")
        self._celsius = value

    @property
    def fahrenheit(self):
        return (self._celsius * 9/5) + 32

    @fahrenheit.setter
    def fahrenheit(self, value):
        self.celsius = (value - 32) * 5/9

# Using properties
temp = Temperature(25)
print(f"{temp.celsius}°C = {temp.fahrenheit}°F")

temp.celsius = 0
print(f"{temp.celsius}°C = {temp.fahrenheit}°F")

temp.fahrenheit = 100
print(f"{temp.celsius}°C = {temp.fahrenheit}°F")

# This would raise an error:
# temp.celsius = -300  # ValueError!
