9.2 CRUD Operations

CRUD stands for Create, Read, Update, Delete - the fundamental database operations.

import sqlite3

class StudentDatabase:
    def __init__(self, db_name='students.db'):
        self.conn = sqlite3.connect(db_name)
        self.cursor = self.conn.cursor()
        self.create_table()

    def create_table(self):
        self.cursor.execute('''
            CREATE TABLE IF NOT EXISTS students (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                age INTEGER,
                grade REAL
            )
        ''')
        self.conn.commit()

    def create(self, name, age, grade):
        """Create - Insert new student"""
        self.cursor.execute(
            'INSERT INTO students (name, age, grade) VALUES (?, ?, ?)',
            (name, age, grade)
        )
        self.conn.commit()
        print(f"Added {name} to database")

    def read_all(self):
        """Read - Get all students"""
        self.cursor.execute('SELECT * FROM students')
        return self.cursor.fetchall()

    def read_one(self, student_id):
        """Read - Get one student"""
        self.cursor.execute('SELECT * FROM students WHERE id = ?', (student_id,))
        return self.cursor.fetchone()

    def update(self, student_id, name=None, age=None, grade=None):
        """Update - Modify student data"""
        if name:
            self.cursor.execute(
                'UPDATE students SET name = ? WHERE id = ?',
                (name, student_id)
            )
        if age:
            self.cursor.execute(
                'UPDATE students SET age = ? WHERE id = ?',
                (age, student_id)
            )
        if grade:
            self.cursor.execute(
                'UPDATE students SET grade = ? WHERE id = ?',
                (grade, student_id)
            )
        self.conn.commit()
        print(f"Updated student ID {student_id}")

    def delete(self, student_id):
        """Delete - Remove student"""
        self.cursor.execute('DELETE FROM students WHERE id = ?', (student_id,))
        self.conn.commit()
        print(f"Deleted student ID {student_id}")

    def close(self):
        self.conn.close()

# Usage
db = StudentDatabase()
db.create('Alice', 20, 3.8)
db.create('Bob', 22, 3.5)

students = db.read_all()
for student in students:
    print(student)

db.update(1, grade=3.9)
db.delete(2)

db.close()

    __tablename__ = 'students'

    id = Column(Integer, primary_key=True)
    name = Column(String, nullable=False)
    age = Column(Integer)
    grade = Column(Float)

    def __repr__(self):
        return f""

# Create engine and session
engine = create_engine('sqlite:///school.db')
Base.metadata.create_all(engine)
Session = sessionmaker(bind=engine)

11.2 Building a REST API

REST APIs allow applications to communicate over HTTP. Let's build a simple TODO API!

from flask import Flask, jsonify, request

app = Flask(__name__)

# In-memory database
todos = [
    {'id': 1, 'task': 'Learn Python', 'done': True},
    {'id': 2, 'task': 'Build API', 'done': False}
]

# GET all todos
@app.route('/api/todos', methods=['GET'])
def get_todos():
    return jsonify({'todos': todos})

# GET single todo
@app.route('/api/todos/', methods=['GET'])
def get_todo(todo_id):
    todo = next((t for t in todos if t['id'] == todo_id), None)
    if todo:
        return jsonify(todo)
    return jsonify({'error': 'Todo not found'}), 404

# CREATE todo
@app.route('/api/todos', methods=['POST'])
def create_todo():
    new_todo = {
        'id': len(todos) + 1,
        'task': request.json['task'],
        'done': False
    }
    todos.append(new_todo)
    return jsonify(new_todo), 201

# UPDATE todo
@app.route('/api/todos/', methods=['PUT'])
def update_todo(todo_id):
    todo = next((t for t in todos if t['id'] == todo_id), None)
    if todo:
        todo['task'] = request.json.get('task', todo['task'])
        todo['done'] = request.json.get('done', todo['done'])
        return jsonify(todo)
    return jsonify({'error': 'Todo not found'}), 404

# DELETE todo
@app.route('/api/todos/', methods=['DELETE'])
def delete_todo(todo_id):
    global todos
    todos = [t for t in todos if t['id'] != todo_id]
    return jsonify({'message': 'Todo deleted'})

if __name__ == '__main__':
    app.run(debug=True)

# File operations
def organize_files(source_dir):
    """Organize files by extension"""
    source = Path(source_dir)

    for file in source.iterdir():
        if file.is_file():
            # Get extension
            ext = file.suffix[1:]  # Remove the dot
            if ext:
                # Create folder for extension
                dest_folder = source / ext
                dest_folder.mkdir(exist_ok=True)

                # Move file
                shutil.move(str(file), str(dest_folder / file.name))
                print(f"Moved {file.name} to {ext}/")

# Batch rename files
def batch_rename(directory, old_text, new_text):
    """Rename multiple files"""
    path = Path(directory)
    for file in path.iterdir():
        if file.is_file() and old_text in file.name:
            new_name = file.name.replace(old_text, new_text)
            file.rename(path / new_name)
            print(f"Renamed: {file.name} → {new_name}")

# Search files by content
def search_in_files(directory, search_term):
    """Search for text in files"""
    results = []
    for file in Path(directory).rglob('*.txt'):
        try:
            content = file.read_text()
            if search_term in content:
                results.append(file)
                print(f"Found in: {file}")
        except Exception as e:
            print(f"Error reading {file}: {e}")
    return results

    """Extract data from HTML table"""

    data = []

    for row in table.find_all('tr')[1:]:  # Skip header

        data.append([col.text.strip() for col in cols])

class Robot:
    def __init__(self):
        self.position = {'x': 0, 'y': 0}
        self.direction = 'N'  # North
        print("Robot initialized")

    def move_forward(self, distance):
        if self.direction == 'N':
            self.position['y'] += distance
        elif self.direction == 'S':
            self.position['y'] -= distance
        elif self.direction == 'E':
            self.position['x'] += distance
        elif self.direction == 'W':
            self.position['x'] -= distance
        print(f"Moved forward {distance} units")
        print(f"Position: {self.position}")

    def turn_left(self):
        directions = ['N', 'W', 'S', 'E']
        idx = directions.index(self.direction)
        self.direction = directions[(idx + 1) % 4]
        print(f"Turned left, now facing {self.direction}")

    def turn_right(self):
        directions = ['N', 'E', 'S', 'W']
        idx = directions.index(self.direction)
        self.direction = directions[(idx + 1) % 4]
        print(f"Turned right, now facing {self.direction}")

    def get_status(self):
        return {
            'position': self.position,
            'direction': self.direction
        }

# Create and control robot

robot.turn_right()
